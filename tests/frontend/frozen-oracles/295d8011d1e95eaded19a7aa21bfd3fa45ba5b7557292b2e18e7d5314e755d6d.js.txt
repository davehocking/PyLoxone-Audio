// Optional PyLoxone-Audio card. Uses only the owning HA entity/state/services.
const FEATURES = {volume: 4, pause: 1, play: 16384, grouping: 524288};
// @mdi/js 7.4.47 standard paths, rendered by HA's own ha-svg-icon.
const ICONS = {play:'M8,5.14V19.14L19,12.14L8,5.14Z',pause:'M14,19H18V5H14M6,19H10V5H6V19Z',minus:'M19,13H5V11H19V13Z',plus:'M19,13H13V19H11V13H5V11H11V5H13V11H19V13Z'};
const percentOf = state => {
  const value=state?.attributes?.volume_level;
  return typeof value==='number'&&Number.isFinite(value)&&value>=0&&value<=1?Math.round(value*100):null;
};
const entityId = value => typeof value === 'string' && /^media_player\.[a-z0-9_]+$/.test(value);
const same = (a,b) => JSON.stringify(a) === JSON.stringify(b);
const nameOf = state => state.attributes.friendly_name || state.entity_id;
const compareNames = (hass,a,b) => {
  const x=nameOf(hass.states[a]).toLocaleLowerCase('en'),y=nameOf(hass.states[b]).toLocaleLowerCase('en');
  return x<y?-1:x>y?1:a<b?-1:a>b?1:0;
};
const contexts = hass => Object.keys(hass?.states||{}).filter(id=>project(hass,id)).sort((a,b)=>compareNames(hass,a,b));
const transportAction = state => {
  if(!state||!['playing','paused','idle','off','on','standby'].includes(state.state))return null;
  const action=state.state==='playing'?'pause':'play';
  return state.attributes.supported_features&FEATURES[action]?action:null;
};
const node = (tag, text, attrs={}) => {
  const element=document.createElement(tag); if(text!==null)element.textContent=text;
  for(const [key,value] of Object.entries(attrs))element.setAttribute(key,value);
  return element;
};

export function project(hass, target) {
  const state=hass?.states?.[target], d=state?.attributes?.loxone_group_control;
  const available=s=>s && !['unknown','unavailable'].includes(s.state);
  const ids=list=>Array.isArray(list)&&list.length>0&&list.length<=256&&list.every(entityId)&&new Set(list).size===list.length;
  if(!available(state)||!d||d.version!==1||typeof d.owner!=='string'||typeof d.revision!=='string'
    ||!ids(d.members)||!ids(d.candidates)||!d.members.includes(target)||!same(d.members,state.attributes.group_members))return null;
  const candidates=d.candidates.filter(id=>{
    const s=hass.states[id], other=s?.attributes?.loxone_group_control;
    return available(s)&&hass.entities?.[id]?.platform==='loxone'&&other?.owner===d.owner
      &&other.version===1&&other.revision===d.revision&&other.candidates?.includes(id)
      &&!!(s.attributes.supported_features&FEATURES.grouping);
  });
  if(!d.members.every(id=>candidates.includes(id)))return null;
  const compare=(a,b)=>compareNames(hass,a,b);
  return {owner:d.owner,revision:d.revision,members:[...d.members],leader:d.members[0],group:d.group_id,
    busy:!!d.busy,candidates,top:candidates.filter(id=>d.members.includes(id)).sort(compare),
    other:candidates.filter(id=>!d.members.includes(id)).sort(compare)};
}

        throw new Error('Native membership changed externally; no further commands sent.');
      if(nudge&&(!same(now.members,original.members)||!nudge.row.isConnected||this._rows.get(nudge.id)!==nudge.row
        ||percentOf(this.hass.states[nudge.id])===null||!(this.hass.states[nudge.id]?.attributes.supported_features&FEATURES.volume)))
        throw new Error('Room volume capability or native membership changed; no further commands sent.');
      if(Date.now()>=deadline)throw new Error('Membership or request not confirmed by native feedback; no retry or rollback sent.');
    };
    const guard=new Promise((_,reject)=>{
      token.reject=reject;
      timer=setInterval(()=>{
        try{ensure();}catch(error){reject(error);}
      },40);
    });
    try{
      await Promise.race([guard,(async()=>{
        await work();ensure();
        if(expected)while(this._token===token&&!same(project(this.hass,entity)?.members,expected))await new Promise(resolve=>setTimeout(resolve,40));
        if(nudge)while(this._token===token&&percentOf(this.hass.states[nudge.id])!==nudge.percent)await new Promise(resolve=>setTimeout(resolve,40));
        ensure();
      })()]);
      if(this._token===token)this._notice='Request accepted. Displayed values follow native feedback.';
    }
    catch(error){if(this._token===token)this._error=error?.message||'Native operation failed; refresh current state before retrying.';}
    finally{clearInterval(timer);if(this._token===token){this._token=null;this._pending=false;this._render();}}
  }
  _volume(id,value,row){
    const view=project(this.hass,this._entity),state=this.hass?.states?.[id];
    if(row&&(!row.isConnected||this._rows.get(id)!==row||row.dataset.owner!==view?.owner))return;
    if(!view||view.busy||!view.candidates.includes(id)||!(state.attributes.supported_features&FEATURES.volume)||!Number.isInteger(value)||value<0||value>100)return;
    return this._run(()=>this.hass.callService('media_player','volume_set',{volume_level:value/100},{entity_id:id}));
  }
  _nudge(id,direction,row){
    const view=project(this.hass,this._entity),state=this.hass?.states?.[id],percent=percentOf(state);
    if(!this._valid(this._stamp(view),id,row)||percent===null||![-1,1].includes(direction)
      ||!(state.attributes.supported_features&FEATURES.volume))return;
    const requested=Math.min(100,Math.max(0,percent+direction));if(requested===percent)return;
    return this._run(()=>this.hass.callService('media_player','volume_set',{volume_level:requested/100},{entity_id:id}),null,{id,percent:requested,row});
  }
  _roomPlay(id,row){
    const view=project(this.hass,this._entity),state=this.hass?.states?.[id];
    if(!row.isConnected||this._rows.get(id)!==row||!view||view.busy||row.dataset.owner!==view.owner||!view.candidates.includes(id))return;
    const action=transportAction(state);if(!action)return;
    return this._run(()=>this.hass.callService('loxone','audio_room_transport',{action},{entity_id:id}));
  }
  _groupPlay(){
    const view=project(this.hass,this._entity);if(!view||view.busy)return;
    const action=transportAction(this.hass.states[view.leader]);if(!action)return;
    const service='media_'+action;
    return this._run(()=>this.hass.callService('media_player',service,{}, {entity_id:view.leader}));
  }
  async _membership(id,included,row){
    const original=project(this.hass,this._entity);if(!original||original.busy)return;
    if(!row.isConnected||this._rows.get(id)!==row||row.dataset.owner!==original.owner||!original.candidates.includes(id)||id===this._entity||this._pending){this._render();return;}
    if(included===original.members.includes(id)){this._render();return;}
    const expected=included?[...original.members,id]:original.members.filter(x=>x!==id);
    return this._run(()=>included
      ?this.hass.callService('media_player','join',{group_members:[id]},{entity_id:this._entity})
      :this.hass.callService('media_player','unjoin',{}, {entity_id:id}),expected);
  }
}

class LoxoneGroupControlEditor extends HTMLElement {
  constructor(){super();this.attachShadow({mode:'open'});}
  setConfig(config){this._config={...config};this._render();}
  set hass(value){this._hass=value;this._render();}
  _render(){
    if(!this._hass||!this._config)return;
    this.shadowRoot.replaceChildren();const label=node('label','Loxone speaker '),select=node('select',null,{'aria-label':'Loxone speaker'});
    for(const id of Object.keys(this._hass.states).filter(id=>project(this._hass,id)).sort()){
      const option=node('option',nameOf(this._hass.states[id]),{value:id});select.append(option);
    }
    select.value=this._config.entity;select.onchange=()=>{this._config={...this._config,entity:select.value};this.dispatchEvent(new CustomEvent('config-changed',{detail:{config:this._config},bubbles:true,composed:true}));};
    label.append(select);this.shadowRoot.append(label);
    const presentationLabel=node('label','Presentation '),presentation=node('select',null,{'aria-label':'Presentation'});
    presentation.append(node('option','Groups button',{value:'button'}),node('option','Native media player + Groups',{value:'player'}));
    presentation.value=this._config.presentation||'button';
    presentation.onchange=()=>{this._config={...this._config,presentation:presentation.value};this.dispatchEvent(new CustomEvent('config-changed',{detail:{config:this._config},bubbles:true,composed:true}));};
    presentationLabel.append(presentation);this.shadowRoot.append(presentationLabel);
  }
}
